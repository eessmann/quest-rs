cmake_minimum_required(VERSION 3.28)
project(quest_qsvt_execution_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(REFERENCE_SOURCE "" CACHE PATH "Pinned quest-qsvt checkout")
set(REFERENCE_CPM_BOOTSTRAP "" CACHE FILEPATH "Optional existing CPM 0.42.0 bootstrap for offline configuration")
if(NOT REFERENCE_SOURCE)
  message(FATAL_ERROR "Set REFERENCE_SOURCE to the pinned quest-qsvt checkout")
endif()
execute_process(COMMAND git -C "${REFERENCE_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE reference_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT reference_revision STREQUAL "7fe7f740579b03c52a8cf48be6a31268b029c19f")
  message(FATAL_ERROR "Execution reference requires pinned revision 7fe7f740579b03c52a8cf48be6a31268b029c19f")
endif()
if(REFERENCE_CPM_BOOTSTRAP)
  file(SHA256 "${REFERENCE_CPM_BOOTSTRAP}" bootstrap_hash)
  if(NOT bootstrap_hash STREQUAL "2020b4fc42dba44817983e06342e682ecfc3d2f484a581f11cc5731fbe4dce8a")
    message(FATAL_ERROR "CPM bootstrap hash mismatch")
  endif()
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/cmake")
  configure_file("${REFERENCE_CPM_BOOTSTRAP}" "${CMAKE_BINARY_DIR}/cmake/CPM_0.42.0.cmake" COPYONLY)
endif()
set(quest_qsvt_ENABLE_DOCS OFF CACHE BOOL "" FORCE)
set(quest_qsvt_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(quest_qsvt_ENABLE_HARDENING OFF CACHE BOOL "" FORCE)
set(quest_qsvt_ENABLE_CACHE OFF CACHE BOOL "" FORCE)
set(quest_qsvt_ENABLE_QUEST ON CACHE BOOL "" FORCE)
set(quest_qsvt_FETCH_DEPS OFF CACHE BOOL "" FORCE)
set(quest_qsvt_FFT_BACKEND "POCKETFFT" CACHE STRING "" FORCE)
find_package(ZLIB REQUIRED)
# The dependency's config package names its static target differently from HDF5.
if(NOT TARGET ZLIB::ZLIB AND TARGET ZLIB::ZLIBSTATIC)
  add_library(ZLIB::ZLIB ALIAS ZLIB::ZLIBSTATIC)
endif()
add_subdirectory("${REFERENCE_SOURCE}" current-source EXCLUDE_FROM_ALL)
add_executable(cpp_execution "${CMAKE_CURRENT_LIST_DIR}/cpp_execution.cpp")
target_link_libraries(cpp_execution PRIVATE quest_qsvt::quest_executor quest_qsvt::transforms quest_qsvt::hadamard glaze::glaze)
target_compile_options(cpp_execution PRIVATE -O3 -march=native -frounding-math -ffp-contract=off)
foreach(adapter IN ITEMS cpp_qsp cpp_gqsp)
  add_executable(${adapter} EXCLUDE_FROM_ALL "${CMAKE_CURRENT_LIST_DIR}/${adapter}.cpp")
  target_link_libraries(${adapter} PRIVATE qsp_solver glaze::glaze)
endforeach()
