cmake_minimum_required(VERSION 3.28)
project(qsp_branch_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(REFERENCE_SOURCE "" CACHE PATH "Verified git archive extraction")
set(REFERENCE_BRANCH "" CACHE STRING "main or develop")
if(REFERENCE_BRANCH STREQUAL "main")
  set(expected_archive "8555d357c8e66d414cae3c0f558e76221b668b9329b4e16a0f39c4081f200cec")
elseif(REFERENCE_BRANCH STREQUAL "develop")
  set(expected_archive "bb1905be1bc5a12f743104bf000c371f0e567169ba3f1df9e3672d248651241e")
else()
  message(FATAL_ERROR "Specify the pinned main or develop reference")
endif()
file(SHA256 "${REFERENCE_SOURCE}/../${REFERENCE_BRANCH}.tar" actual_archive)
if(NOT actual_archive STREQUAL expected_archive)
  message(FATAL_ERROR "Pinned archive SHA256 mismatch")
endif()
# Compile a fresh extraction of the hash-checked immutable archive, never a
# potentially edited source directory supplied by the caller.
set(verified_source "${CMAKE_BINARY_DIR}/verified-source")
file(MAKE_DIRECTORY "${verified_source}")
file(ARCHIVE_EXTRACT INPUT "${REFERENCE_SOURCE}/../${REFERENCE_BRANCH}.tar" DESTINATION "${verified_source}")
set(quest_qsvt_ENABLE_DOCS OFF CACHE BOOL "" FORCE)
set(quest_qsvt_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(quest_qsvt_ENABLE_HARDENING OFF CACHE BOOL "" FORCE)
set(quest_qsvt_ENABLE_CACHE OFF CACHE BOOL "" FORCE)
set(quest_qsvt_ENABLE_QUEST OFF CACHE BOOL "" FORCE)
set(quest_qsvt_FETCH_DEPS OFF CACHE BOOL "" FORCE)
set(quest_qsvt_ENABLE_IO_TESTS OFF CACHE BOOL "" FORCE)
set(quest_qsvt_FFT_BACKEND "POCKETFFT" CACHE STRING "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
# Main requests separately packaged header-only Boost components. The pinned
# develop Nix environment supplies the same headers through Boost::headers.
# This changes dependency target discovery only, never reference source code.
if(REFERENCE_BRANCH STREQUAL "main")
  find_package(Boost CONFIG REQUIRED COMPONENTS headers)
  set(header_configs "${CMAKE_BINARY_DIR}/boost-header-configs")
  file(MAKE_DIRECTORY "${header_configs}")
  foreach(component mp11 hana multiprecision uuid)
    file(WRITE "${header_configs}/boost_${component}Config.cmake"
      "if(NOT TARGET Boost::${component})\nadd_library(Boost::${component} INTERFACE IMPORTED GLOBAL)\nset_target_properties(Boost::${component} PROPERTIES INTERFACE_LINK_LIBRARIES Boost::headers)\nendif()\nset(boost_${component}_FOUND TRUE)\n")
  endforeach()
  list(PREPEND CMAKE_PREFIX_PATH "${header_configs}")
endif()
add_subdirectory("${verified_source}" pinned-source EXCLUDE_FROM_ALL)
add_executable(branch_qsp "${REFERENCE_BRANCH}.cpp")
target_link_libraries(branch_qsp PRIVATE qsp_solver)
target_compile_options(branch_qsp PRIVATE -O2 -frounding-math -ffp-contract=off)
