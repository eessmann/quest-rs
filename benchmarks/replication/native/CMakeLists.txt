cmake_minimum_required(VERSION 3.28)
project(quest_replication_inputs LANGUAGES C CXX)
if(NOT DEFINED QUEST_QSVT_SOURCE)
  message(FATAL_ERROR "Set QUEST_QSVT_SOURCE to the pinned unmodified source checkout")
endif()
execute_process(COMMAND git -C "${QUEST_QSVT_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE source_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT source_revision STREQUAL "4fc35983138d07a990862a4d83ad16f2b737c98f")
  message(FATAL_ERROR "Input exporter requires the pinned quest-qsvt revision")
endif()
set(CMAKE_CXX_STANDARD 23)
set(quest_qsvt_ENABLE_DOCS OFF CACHE BOOL "" FORCE)
add_subdirectory("${QUEST_QSVT_SOURCE}" upstream EXCLUDE_FROM_ALL)
find_package(Catch2 3 CONFIG REQUIRED)
add_executable(export_named export_named.cpp)
target_include_directories(export_named PRIVATE "${QUEST_QSVT_SOURCE}/test/qsp_tools")
target_link_libraries(export_named PRIVATE quest_qsvt::qsp_tools quest_qsvt::transforms Catch2::Catch2)
