cmake_minimum_required(VERSION 3.24)
project(quest_review LANGUAGES CXX)
find_package(QuEST 4.3 REQUIRED CONFIG)
set(QUEST_SOURCE_DIR "" CACHE PATH "Path to the reviewed QuEST source checkout")
if(NOT EXISTS "${QUEST_SOURCE_DIR}/quest/src/api/calculations.cpp")
  message(FATAL_ERROR "Set QUEST_SOURCE_DIR to the reviewed QuEST source checkout")
endif()
get_target_property(mpi_libs MPI::MPI_CXX INTERFACE_LINK_LIBRARIES)
get_target_property(mpi_inc MPI::MPI_CXX INTERFACE_INCLUDE_DIRECTORIES)
message(STATUS "Reviewed MPI libraries: ${mpi_libs}")
message(STATUS "Reviewed MPI includes: ${mpi_inc}")
add_executable(witness witness.cpp "${QUEST_SOURCE_DIR}/quest/src/api/calculations.cpp")
target_compile_features(witness PRIVATE cxx_std_20)
target_include_directories(witness BEFORE PRIVATE
  "$<TARGET_PROPERTY:QuEST::QuEST,INTERFACE_INCLUDE_DIRECTORIES>"
  "${QUEST_SOURCE_DIR}")
target_compile_options(witness PRIVATE -fsanitize=shift -fsanitize-undefined-trap-on-error)
target_link_options(witness PRIVATE -fsanitize=shift -fsanitize-undefined-trap-on-error)
target_link_libraries(witness PRIVATE QuEST::QuEST)
